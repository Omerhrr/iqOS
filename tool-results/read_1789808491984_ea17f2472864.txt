   605→   605→io.on('connection', (socket) => {
   606→   606→  console.log(`[ws] client connected: ${socket.id}`)
   607→   607→  socket.on('subscribe', ({ asset, tf }: { asset?: string; tf?: string }) => {
   608→   608→    for (const room of socket.rooms) {
   609→   609→      if (room.startsWith('feed:')) socket.leave(room)
   610→   610→    }
   611→   611→    socket.join('feed:global')
   612→   612→    if (asset && tf) socket.join(`feed:${asset}:${tf}`)
   613→   613→  })
   614→   614→  socket.on('ping-msg', (data) => socket.emit('pong-msg', data))
   615→   615→  socket.on('disconnect', () => console.log(`[ws] client gone: ${socket.id}`))
   616→   616→})
