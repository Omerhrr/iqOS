    35→const httpServer = createServer(async (req, res) => {
    36→  res.setHeader('access-control-allow-origin', '*')
    37→  res.setHeader('access-control-allow-methods', 'GET,POST,OPTIONS')
    38→  res.setHeader('access-control-allow-headers', 'content-type')
    39→  if (req.method === 'OPTIONS') {
    40→    res.writeHead(204)
    41→    res.end()
    42→    return
    43→  }
    44→
    45→  const url = new URL(req.url ?? '/', `http://127.0.0.1:${PORT}`)
    46→  const path = url.pathname
    47→  const q = url.searchParams
    48→  let body: Record<string, unknown> = {}
    49→  if (req.method === 'POST') {
    50→    const chunks: Uint8Array[] = []
    51→    for await (const chunk of req) chunks.push(chunk as Uint8Array)
    52→    try {
    53→      body = JSON.parse(Buffer.concat(chunks).toString() || '{}') as Record<string, unknown>
    54→    } catch {
    55→      body = {}
    56→    }
    57→  }
    58→
    59→  const json = (code: number, data: unknown) => {
    60→    res.writeHead(code, { 'content-type': 'application/json' })
    61→    res.end(JSON.stringify(data))
    62→  }
    63→  const tf = (name: string | null): Timeframe => {
    64→    const v = (name ?? '1m') as Timeframe
    65→    return (ALL_TIMEFRAMES as string[]).includes(v) ? v : '1m'
    66→  }

    70→    const analytics = kernel.context().use<AnalyticsService>('analytics')
    71→    const exec = kernel.context().use<ExecutionService>('execution')
    72→
    73→    if (req.method === 'GET') {
    74→      if (path === '/health') return json(200, { ok: true, service: 'trading-core', uptime: process.uptime() })
    75→
    76→      if (path === '/assets') return json(200, { ok: true, assets: market.listAssets(), mode: market.mode, activeAsset: market.activeAsset })
    77→
    78→      if (path === '/instruments') {
    79→        const cat = (q.get('category') ?? 'all') as 'all' | 'otc' | 'forex' | 'crypto' | 'commodity' | 'stock' | 'index'
    80→        const search = q.get('q') ?? ''
    81→        const found = searchInstruments(search, cat)
    82→        market.refreshSchedules()

    84→          ok: true,
    85→          instruments: found.map((a) => ({ ...a, price: market.getPrice(a.ticker) || a.basePrice })),

    95→        const id = q.get('id') ?? 'rsi'
    96→        const def = getIndicatorDef(id)
    97→        if (!def) return json(404, { ok: false, error: `unknown indicator ${id}` })
    98→        const asset = q.get('asset') ?? market.activeAsset
    99→        const timeframe = tf(q.get('tf'))
   100→        const params: Record<string, number> = {}
   101→        for (const [k, v] of q.entries()) if (k.startsWith('p_')) params[k.slice(2)] = Number(v)
   102→        const candles = market.getCandles(asset, timeframe, 400)
   103→        const res = computeIndicator(id, candles, params)
   104→        if (!res) return json(404, { ok: false, error: 'compute failed' })

   106→        const nz = (v: number) => (Number.isFinite(v) ? v : null)

   108→          ok: true,
   109→          series: {
   110→            id: def.id,
   111→            name: def.name,
   112→            category: def.category,
   113→            pane: def.pane,
   114→            params: Object.fromEntries(def.params.map((pp) => [pp.key, params[pp.key] ?? pp.default])),
   115→            time,
   116→            lines: res.output.lines.map((ln) => ({ key: ln.key, color: ln.color, style: ln.style, values: ln.values.map(nz) })),

   118→            levels: res.output.levels,
   119→            bands: res.output.bands,

   122→          },
   123→        })

   149→        const cat = (q.get('category') ?? 'all') as 'all' | 'otc' | 'forex' | 'crypto' | 'commodity' | 'stock' | 'index'
   150→        const timeframe = tf(q.get('tf'))
   151→        const cap = Math.min(Number(q.get('limit') ?? 14), 24)
   152→        const universe = searchInstruments('', cat)
   153→          .filter((i) => i.open)
   154→          .slice(0, cap)

   156→        const CHUNK = 5
   157→        for (let i = 0; i < universe.length; i += CHUNK) {
   158→          await Promise.all(
   159→            universe.slice(i, i + CHUNK).map(async (inst) => {
   160→              try {
   161→                const a = analytics.analyze(inst.ticker, timeframe)
   162→                rows.push({
   163→                  asset: inst.ticker,
   164→                  name: inst.name,
   165→                  category: inst.category,
   166→                  price: a.signal.price,
   167→                  score: Math.round(a.signal.score * 10) / 10,
   168→                  direction: a.signal.direction,
   169→                  confidence: Math.round(a.signal.confidence),
   170→                  pUp: Math.round(a.markov.probUp * 1000) / 1000,
   171→                  regime: a.markov.regime,
   172→                  rsi: Math.round(a.indicators.rsi * 10) / 10,
   173→                  hurst: Math.round(a.quant.hurst * 100) / 100,
   174→                })
   175→              } catch {
   176→                // insufficient candles - skip
   177→              }
   178→            })
   179→          )
   180→        }
   181→        rows.sort((x, y) => Math.abs(Number(y.score)) - Math.abs(Number(x.score)))
   182→        return json(200, { ok: true, tf: timeframe, scanned: universe.length, results: rows })
   183→      }

   282→        const scope = q.get('scope') ?? 'all' // all | bots
   283→        const trades = store.journal(scope === 'bots' ? 'bot:' : undefined, 500) as {
   284→          tsOpen: number
   285→          tsClose?: number
   286→          asset: string
   287→          side: string
   288→          kind: string
   289→          strategy?: string
   290→          amount: number
   291→          pnl?: number
   292→          status: string
   293→        }[]
   294→        const closed = trades.filter((t) => t.status === 'won' || t.status === 'lost')
   295→        const wins = closed.filter((t) => t.status === 'won')
   296→        const losses = closed.filter((t) => t.status === 'lost')
   297→        const pnls = closed.map((t) => t.pnl ?? 0)
   298→        const netPnl = pnls.reduce((a, b) => a + b, 0)
   299→        const grossWin = pnls.filter((p) => p > 0).reduce((a, b) => a + b, 0)
   300→        const grossLoss = Math.abs(pnls.filter((p) => p < 0).reduce((a, b) => a + b, 0))
   301→        const group = (keyOf: (t: (typeof closed)[number]) => string) => {
   302→          const map = new Map<string, { trades: number; wins: number; pnl: number }>()
   303→          for (const t of closed) {

   305→            const g = map.get(k) ?? { trades: 0, wins: 0, pnl: 0 }
   306→            g.trades += 1
   307→            if (t.status === 'won') g.wins += 1
   308→            g.pnl += t.pnl ?? 0
   309→            map.set(k, g)
   310→          }
   311→          return [...map.entries()]
   312→            .map(([key, g]) => ({ key, ...g, winRate: g.trades ? g.wins / g.trades : 0 }))
   313→            .sort((a, b) => b.pnl - a.pnl)
   314→        }
   315→        // chronological equity curve

   333→            profitFactor: grossLoss > 0 ? Math.round((grossWin / grossLoss) * 100) / 100 : grossWin > 0 ? 99 : 0,
   334→            bestTrade: pnls.length ? Math.round(Math.max(...pnls) * 100) / 100 : 0,
   335→            worstTrade: pnls.length ? Math.round(Math.min(...pnls) * 100) / 100 : 0,
   336→            avgWin: wins.length ? Math.round((grossWin / wins.length) * 100) / 100 : 0,
   337→            avgLoss: losses.length ? Math.round((grossLoss / losses.length) * 100) / 100 : 0,
   338→          },
   339→          curve,
   340→          byStrategy: group((t) => t.strategy ?? ''),
   341→          byAsset: group((t) => t.asset),
   342→          byKind: group((t) => t.kind),
   343→          bySide: group((t) => t.side),

   450→        const inst = getInstrument(asset)
   451→        if (!inst) return json(400, { ok: false, error: `unknown asset ${asset}` })
   452→        const out = await exec.placeOrder({
   453→          asset,
   454→          tf: tf(String(body.tf ?? '1m') as string),
   455→          side: (body.side as 'call' | 'put') ?? 'call',
   456→          kind,
   457→          amount: Number(body.amount ?? 10),
   458→          expiryBars: body.expiryBars !== undefined ? Number(body.expiryBars) : undefined,
   459→          expirySec: body.expirySec !== undefined ? Number(body.expirySec) : undefined,
   460→          strikeOffsetPct: body.strikeOffsetPct !== undefined ? Number(body.strikeOffsetPct) : undefined,
   461→          mode: (body.mode as 'paper' | 'live') ?? 'paper',
   462→          tp: body.tp !== undefined ? Number(body.tp) : undefined,
   463→          sl: body.sl !== undefined ? Number(body.sl) : undefined,
   464→          leverage: body.leverage !== undefined ? Number(body.leverage) : undefined,
   465→          strategy: body.strategy !== undefined ? String(body.strategy) : undefined,
   466→          note: body.note !== undefined ? String(body.note) : undefined,
   467→        })
   468→        return json(200, out)
   469→      }

   471→      if (path === '/close') {

   597→const io = new Server(httpServer, {
   598→  // default path '/socket.io/' so only engine.io requests are claimed;
   599→  // plain REST requests fall through to the handler above.
   600→  cors: { origin: '*', methods: ['GET', 'POST'] },
   601→  pingTimeout: 60000,
   602→  pingInterval: 25000,
   603→})
   604→
   605→io.on('connection', (socket) => {
   606→  console.log(`[ws] client connected: ${socket.id}`)
   607→  socket.on('subscribe', ({ asset, tf }: { asset?: string; tf?: string }) => {
   608→    for (const room of socket.rooms) {
   609→      if (room.startsWith('feed:')) socket.leave(room)
   610→    }
   611→    socket.join('feed:global')
   612→    if (asset && tf) socket.join(`feed:${asset}:${tf}`)
   613→  })
   614→  socket.on('ping-msg', (data) => socket.emit('pong-msg', data))
   615→  socket.on('disconnect', () => console.log(`[ws] client gone: ${socket.id}`))
   616→})
   617→
   618→// forward kernel events: candles room-scoped by asset+tf, everything else global
   619→const ctx = kernel.context()
   620→ctx.bus.on('tick', (p) => io.to('feed:global').emit('tick', p))
   621→ctx.bus.on('candle', (p) => io.to(`feed:${p.asset}:${p.tf}`).emit('candle', p))
   622→ctx.bus.on('positionOpened', (p) => io.to('feed:global').emit('positionOpened', p))
   623→ctx.bus.on('positionClosed', (p) => io.to('feed:global').emit('positionClosed', p))
   624→ctx.bus.on('account', (p) => io.to('feed:global').emit('account', p))
   625→ctx.bus.on('alert', (p) => {
   626→  const store = ctx.use<{ recordAlert: (l: string, m: string, t: number) => void }>('storeRaw')
   627→  store.recordAlert(p.level, p.message, p.ts)
   628→  io.to('feed:global').emit('alert', p)
   629→})
